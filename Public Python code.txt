coords3 = np.column_stack([
    np.cos(lat) * np.cos(lon),
    np.cos(lat) * np.sin(lon),

# === A2. FUNKTIOT (version-agnostic SDM impacts + CI) ===
def parse_coeffs(model, X_vars):
    names = model.name_x
    betas = np.asarray(model.betas).flatten()
    if betas.size == len(names) + 1:
        betas_wo_rho = betas[1:]
    else:
        betas_wo_rho = betas.copy()
    beta_map, theta_map = {}, {}
    const_beta = None
    for nm, b in zip(names, betas_wo_rho):
        if nm.upper() in ("CONSTANT","CONST","INTERCEPT"):
            const_beta = float(b); continue
        if nm.startswith("W_"):
            theta_map[nm[2:]] = float(b)
        else:
           beta_map[nm] = float(b)
    for x in X_vars:
        theta_map.setdefault(x, 0.0)
    order_info = [(x, theta_map[x]!=0.0) for x in X_vars]
    return beta_map, theta_map, const_beta, order_info

def impacts_for_var(rho, W, beta_j, theta_j, R=256, rng=None):
    if rng is None:
        rng = default_rng(42)
    n = W.n
    I = speye(n, format="csc")
    Ws = W.sparse.tocsc()
    A = (I - rho * Ws)
    lu = splu(A)
    ones = np.ones((n,1))
    M1 = beta_j*ones + theta_j*(Ws @ ones)
    SM1 = lu.solve(M1)
    total = float(SM1.mean())
    z = rng.choice([-1.0, 1.0], size=(n, R))
    Sz = lu.solve(z)
    trace_S_hat = float((z * Sz).sum(axis=0).mean())
    Wz = Ws @ z
    SWz = lu.solve(Wz)
    trace_SW_hat = float((z * SWz).sum(axis=0).mean())
    direct = (beta_j*trace_S_hat + theta_j*trace_SW_hat) / n
    indirect = total - direct
    return direct, indirect, total

def align_mean_cov(model):
    cov = getattr(model, "vm", None)
    if cov is None:
        cov1 = np.asarray(model.vm1)
        cov = cov1[:-1,:-1]
    else:
        cov = np.asarray(cov)
    names = list(model.name_x)
    bet = np.asarray(model.betas).flatten()
    rho = float(model.rho)
    p = cov.shape[0]
    if bet.size == p:
        if p == len(names)+1:
            theta_hat = np.concatenate(([rho], bet))
            param_names = ["rho"] + names
        else:
            theta_hat = bet
            param_names = names
    elif bet.size == p - 1:
        theta_hat = np.concatenate(([rho], bet))
        param_names = ["rho"] + names
    elif bet.size == p + 1:
        theta_hat = bet[:-1]
        if p == len(names)+1:
            param_names = ["rho"] + names
        else:
            param_names = names
    else:
        raise ValueError(f"Cannot align mean len={bet.size} with cov {cov.shape}")
    cov = (cov + cov.T)/2.0
    cov += 1e-12*np.eye(cov.shape[0])
    return theta_hat, cov, param_names

def build_index_maps(param_names, X_vars):
    name_to_idx = {nm:i for i,nm in enumerate(param_names)}
    rho_idx = name_to_idx.get("rho", None)
    beta_idx = {x: name_to_idx.get(x, None) for x in X_vars}
    theta_idx = {x: name_to_idx.get(f"W_{x}", None) for x in X_vars}
    return rho_idx, beta_idx, theta_idx

def sdm_impacts_with_ci(model, W, X_vars, M=2000, R=256, seed=123):
    rng = default_rng(seed)
    theta_hat, cov, param_names = align_mean_cov(model)
    rho_idx, beta_idx, theta_idx = build_index_maps(param_names, X_vars)
    rho_hat = float(theta_hat[rho_idx]) if rho_idx is not None else float(model.rho)
    tmp = parse_coeffs(model, X_vars)
    if len(tmp) == 4:
        beta_map, theta_map, const_beta, _ = tmp
    else:
        beta_map, theta_map, const_beta = tmp
    # Piste-estimaatit
    point_records=[]
    for x in X_vars:
        d0,i0,t0 = impacts_for_var(rho_hat, W, beta_map.get(x,0.0), theta_map.get(x,0.0), R=R, rng=rng)
        point_records.append((x,d0,i0,t0))
    point_df = pd.DataFrame(point_records, columns=["Variable","Direct","Indirect","Total"])
    # Monte Carlo
    p = cov.shape[0]
    L = np.linalg.cholesky(cov)
    z = rng.standard_normal(size=(M,p))
    draws = theta_hat + z @ L.T
    d_sims=[]; i_sims=[]; t_sims=[]
    for b in draws:
        rho_s = float(b[rho_idx]) if rho_idx is not None else rho_hat
        b_map = {x: (float(b[beta_idx[x]]) if beta_idx[x] is not None else beta_map.get(x,0.0)) for x in X_vars}
        t_map = {x: (float(b[theta_idx[x]]) if theta_idx[x] is not None else theta_map.get(x,0.0)) for x in X_vars}
        d_row=[]; i_row=[]; t_row=[]
        for x in X_vars:
            d,i,t = impacts_for_var(rho_s, W, b_map[x], t_map[x], R=R, rng=rng)
            d_row.append(d); i_row.append(i); t_row.append(t)
        d_sims.append(d_row); i_sims.append(i_row); t_sims.append(t_row)
    d_sims = np.asarray(d_sims); i_sims = np.asarray(i_sims); t_sims = np.asarray(t_sims)
    def ci95(a): return np.percentile(a, 2.5, axis=0), np.percentile(a, 97.5, axis=0)
    d_lo,d_hi = ci95(d_sims); i_lo,i_hi = ci95(i_sims); t_lo,t_hi = ci95(t_sims)
    out = point_df.copy()
    out["Direct_CI_L"]=d_lo; out["Direct_CI_U"]=d_hi
    out["Indirect_CI_L"]=i_lo; out["Indirect_CI_U"]=i_hi
    out["Total_CI_L"]=t_lo; out["Total_CI_U"]=t_hi
    return out

import numpy as np
import pandas as pd

def extract_params(model, model_name):
    """
    Robust extraction of coefficients, std.err, z-scores, p-values
    from spreg.ML_Lag objects, compatible across PySAL versions.
    Returns DataFrame with:
       Variable | Coef | StdErr | z | p | Model
    """

    # --- 1. Model parameter names (including WX terms) ---
    names = list(model.name_x)

    # --- 2. Coefficients, StdErrs directly ---
    coef = np.asarray(model.betas).flatten()
    stderr = np.asarray(model.std_err).flatten()

    # --- 3. z-stat and p-values ---
    # model.z_stat is a list-of-lists: [coef, z, p]
    z = np.array([row[1] for row in model.z_stat])
    p = np.array([row[2] for row in model.z_stat])

    # --- 4. Add rho manually if not included ---
    # Some spreg versions keep rho separate from betas
    if hasattr(model, "rho") and "W_y" not in names:
        names = ["W_y"] + names
        coef = np.insert(coef, 0, model.rho)
        stderr = np.insert(stderr, 0, model.std_err[0])
        z = np.insert(z, 0, model.z_stat[0][1])
        p = np.insert(p, 0, model.z_stat[0][2])

    # --- Build DF ---
    df = pd.DataFrame({
        "Variable": names,
        f"{model_name}_Coef": coef,
        f"{model_name}_StdErr": stderr,
        f"{model_name}_z": z,
        f"{model_name}_p": p
    })

    return df



import numpy as np
import pandas as pd

def extract_params(model, model_name):
    """
    Version-robust extraction of coefficient table from spreg.ML_Lag.
    Returns: DataFrame with columns:
      Variable | <name>_Coef | <name>_StdErr | <name>_z | <name>_p
    Notes:
      - Handles cases where z_stat has only (coef, z) and no p-value.
      - Adds rho as 'W_y' if not already in name_x.
      - Prints diagnostics if vector lengths mismatch.
    """

    # ---- 1) Core arrays ----
    names = list(model.name_x)                     # regressors (includes WX, constant label if present)
    betas = np.asarray(model.betas).flatten()      # coefficients (no rho in most spreg builds)
    se    = np.asarray(model.std_err).flatten()    # std errors (same count as betas + possibly rho as first? varies)
    zraw  = model.z_stat                           # list-like, rows are [coef, z] or [coef, z, p]
    # Normalize z and p safely
    z = []
    p = []
    for row in zraw:
        # Some versions give tuples like (coef, z) or (coef, z, p)
        try:
            # ensure indexable
            r = list(row)
        except Exception:
            r = [np.nan, np.nan, np.nan]
        # z at index 1 if present
        z_val = r[1] if len(r) >= 2 else np.nan
        # p at index 2 if present
        p_val = r[2] if len(r) >= 3 else np.nan
        z.append(z_val)
        p.append(p_val)
    z = np.asarray(z).flatten()
    p = np.asarray(p).flatten()

    # ---- 2) Decide whether rho (W_y) is already inside or must be prepended ----
    # In many spreg versions, betas excludes rho and z_stat includes an extra first row for rho
    # Heuristics:
    #   - If model has attribute 'rho' and 'W_y' not in names, we will prepend it.
    #   - We also try to align se/z/p lengths when prepending.
    add_rho = hasattr(model, "rho") and ("W_y" not in names)

    # Build aligned lists for names / coef / se / z / p (without rho yet)
    coef = betas.copy()
    std  = se.copy()
    zvec = z.copy()
    pvec = p.copy()

    # If z_stat length == betas length + 1, assume first row corresponds to rho
    rho_from_z = None
    if add_rho and (len(zvec) == len(coef) + 1):
        rho_from_z = 0  # index 0 is rho row in z_stat arrays

    # Some builds also put std_err for rho as the first element; we guard against length mismatch.
    if add_rho:
        # Rho coefficient
        rho_coef = float(model.rho)
        # Try to fetch rho se/z/p if we detected a separate row
        if rho_from_z is not None:
            rho_z  = float(zvec[0]) if len(zvec) > 0 else np.nan
            rho_p  = float(pvec[0]) if len(pvec) > 0 else np.nan
            # std err for rho is not always exposed cleanly; try to take the first if lengths align, else NaN
            rho_se = float(std[0]) if len(std) == (len(coef) + 1) else np.nan
            # Now drop that first entry from z/p/se vectors so lengths match betas
            zvec = zvec[1:] if len(zvec) > 0 else zvec
            pvec = pvec[1:] if len(pvec) > 0 else pvec
            if len(std) == (len(coef) + 1):
                std = std[1:]
        else:
            # Could not detect dedicated rho row; set to NaN where unknown
            rho_se = np.nan
            rho_z  = np.nan
            rho_p  = np.nan

        # Prepend to lists
        names = ["W_y"] + names
        coef  = np.insert(coef, 0, rho_coef)
        std   = np.insert(std,  0, rho_se)
        zvec  = np.insert(zvec, 0, rho_z)
        pvec  = np.insert(pvec, 0, rho_p)

    # ---- 3) Basic sanity checks (helpful to debug once, then disable) ----
    if not (len(names) == len(coef) == len(std) == len(zvec) == len(pvec)):
        print(f"[WARN] Length mismatch in {model_name}: "
              f"names={len(names)}, coef={len(coef)}, se={len(std)}, z={len(zvec)}, p={len(pvec)}")
        # Attempt to truncate/pad to the shortest common length
        L = min(len(names), len(coef), len(std), len(zvec), len(pvec))
        names = names[:L]; coef = coef[:L]; std = std[:L]; zvec = zvec[:L]; pvec = pvec[:L]

    # ---- 4) Build DataFrame ----
    df = pd.DataFrame({
        "Variable": names,
        f"{model_name}_Coef":   coef.astype(float),
        f"{model_name}_StdErr": std.astype(float),
        f"{model_name}_z":      zvec.astype(float),
        f"{model_name}_p":      pvec.astype(float)
    })
    return df

def build_dynamic_sdm(df_bal, W_panel, X_vars):
    """
    Builds dynamic panel: Δy, Δy_lag, WΔy, ΔX, WΔX.
    Returns: df_dyn (still includes 1 NA/country), Wp_dyn (comparable), rows.
    """
    import numpy as np
    import pandas as pd

    # 1) Make sure (Year, Country) -order and build row_id
    df_bal = df_bal.sort_values(["Year", "Country"]).reset_index(drop=True)
    df_bal = df_bal.drop(columns=[c for c in df_bal.columns if c == "row_id"], errors="ignore")
    df_bal["row_id"] = np.arange(df_bal.shape[0], dtype=np.int64)

    # 2) Choose columns and make y_{t-1}
    cols = ["Country", "Year", "row_id", "Deforestation_Index"] + X_vars
    df_dyn = df_bal[cols].copy()
    df_dyn["y_lag"] = df_dyn.groupby("Country")["Deforestation_Index"].shift(1)

    # Drop only those that are missing y_lag (first year per country)
    df_dyn = df_dyn.dropna(subset=["y_lag"]).reset_index(drop=True)

    # 3) Put panel into order (Country-major, Year-minor)
    df_dyn = df_dyn.sort_values(["Country", "Year"]).reset_index(drop=True)

    # 4) Δ‑variables
    def diff_by_country(df, col):
        return df.groupby("Country")[col].diff()

    df_dyn["dy"]     = diff_by_country(df_dyn, "Deforestation_Index")
    df_dyn["dy_lag"] = diff_by_country(df_dyn, "y_lag")

    for v in X_vars:
        df_dyn[f"d{v}"] = diff_by_country(df_dyn, v)

    # 5) Cut W_panel for dynamic rows
    rows = df_dyn["row_id"].to_numpy().astype(np.int64)
    W_all = W_panel.sparse.tocsr()
    Wp_dyn = W_all[rows][:, rows]

    # 6) Spatial lags for differences
    df_dyn["dWy"]     = Wp_dyn @ df_dyn["dy"].to_numpy()
    df_dyn["dWy_lag"] = Wp_dyn @ df_dyn["dy_lag"].to_numpy()
    for v in X_vars:
        df_dyn[f"dW{v}"] = Wp_dyn @ df_dyn[f"d{v}"].to_numpy()

    return df_dyn, Wp_dyn, rows



def clean_dyn_for_estimation(df_dyn, Wp_dyn, X_vars):
    """
    Poistaa NA-rivit kaikista Δ- ja WΔ-sarakkeista ja leikkaa W:n samalle maskille.
    Palauttaa: df_cl, W_cl
    """
    import numpy as np

    drop_vars = (
        ["dy","dy_lag","dWy","dWy_lag"]
        + [f"d{v}"  for v in X_vars]
        + [f"dW{v}" for v in X_vars]
    )
    mask_noNA = ~df_dyn[drop_vars].isna().any(axis=1)

    df_cl = df_dyn.loc[mask_noNA].reset_index(drop=True)
    W_cl  = Wp_dyn[mask_noNA.to_numpy(dtype=bool)][:, mask_noNA.to_numpy(dtype=bool)]

    # Pieni sanity printti: montako riviä jäi
    n_before, n_after = df_dyn.shape[0], df_cl.shape[0]
    n_countries = df_dyn["Country"].nunique()
    print(f"[CLEAN] dropped {n_before - n_after} rows (expected ≈ number of countries = {n_countries})")
    return df_cl, W_cl



def run_dynamic_sdm(df_dyn_clean, Wp_dyn_clean, X_vars):
    """
    Δ‑SDM GMM ilman AB-maskin ongelmia:
      y = Δy, X = [Δy_{t-1}, WΔy, ΔX, WΔX], instrumentit = ΔX
    Palauttaa: dyn_gmm (spreg.GM_Lag), SR- ja LR‑vaikutustaulukot.
    """
    import numpy as np
    import pandas as pd
    from spreg import GM_Lag
    import scipy.sparse as sp
    from scipy.sparse.linalg import splu

    # 1) Matriisit
    dy_vec     = df_dyn_clean["dy"].to_numpy().reshape(-1,1)
    dy_lag_vec = df_dyn_clean["dy_lag"].to_numpy().reshape(-1,1)
    dWy_vec    = df_dyn_clean["dWy"].to_numpy().reshape(-1,1)

    dX  = np.column_stack([df_dyn_clean[f"d{v}"].to_numpy()  for v in X_vars])
    dWX = np.column_stack([df_dyn_clean[f"dW{v}"].to_numpy() for v in X_vars])

    Xmat  = np.column_stack([dy_lag_vec, dWy_vec, dX, dWX])
    instr = dX.copy()  # ΔX instrumentteina (vakaa ja NA‑vapaa tässä kohtaa)

    # 2) GMM-estimation
    dyn_gmm = GM_Lag(
        y = dy_vec,
        x = Xmat,
        w = Wp_dyn_clean,
        instruments = instr,
        lag_q = 1,
        w_lags = 0,
        name_y = "dy",
        name_x = ["dy_lag","dWy"] + [f"d{v}" for v in X_vars] + [f"dW{v}" for v in X_vars],
    )

    # 3) Parametrien purku
    params = dyn_gmm.betas.flatten()
    k = len(X_vars)
    phi   = params[0]            # pysyvyys
    rho   = params[1]            # spatiaalinen palaute
    beta  = params[2:2+k]
    theta = params[2+k:2+2*k]

    # 4) SR‑vaikutukset: (I − ρW)^{-1}(β + θW)
    n = Wp_dyn_clean.shape[0]
    I = sp.eye(n, format="csr")
    A = (I - rho * Wp_dyn_clean)
    A_inv = splu(A)

    # Hutchinsonin jälkiarvio tr(S), tr(SW)
    R = 200
    rng = np.random.default_rng(123)
    z = rng.choice([-1.0, 1.0], size=(n, R))
    trace_Ainv  = (z * A_inv.solve(z)).sum(axis=0).mean()
    Wz          = Wp_dyn_clean @ z
    trace_AinvW = (z * A_inv.solve(Wz)).sum(axis=0).mean()

    sr_direct, sr_indirect, sr_total = [], [], []
    ones = np.ones((n,1))
    for j in range(k):
        b, t = beta[j], theta[j]
        M1  = b*ones + t*(Wp_dyn_clean @ ones)
        SM1 = A_inv.solve(M1)
        tot = float(SM1.mean())
        direct   = (b*trace_Ainv + t*trace_AinvW) / n
        indirect = tot - direct
        sr_direct.append(direct); sr_indirect.append(indirect); sr_total.append(tot)

    sr = pd.DataFrame({
        "Variable": X_vars,
        "SR_direct": sr_direct,
        "SR_indirect": sr_indirect,
        "SR_total": sr_total
    })

    # 5) LR‑vaikutukset: SR / (1 − φ)
    factor = 1.0/(1.0 - phi)
    lr = sr.copy()
    lr["LR_direct"]   = lr["SR_direct"]   * factor
    lr["LR_indirect"] = lr["SR_indirect"] * factor
    lr["LR_total"]    = lr["SR_total"]    * factor

    return dyn_gmm, sr, lr


# 1) Rakenna dynaaminen paneeli
df_dyn, Wp_dyn, rows = build_dynamic_sdm(df_bal, W_panel, X_vars)

# 2) Puhdista NA-rivit (≈ 1 per maa) ja leikkaa W samalle joukolle
df_dyn_clean, Wp_dyn_clean = clean_dyn_for_estimation(df_dyn, Wp_dyn, X_vars)

# 3) Estimoi dyn. SDM ja laske SR/LR
dyn_gmm, sr_effects, lr_effects = run_dynamic_sdm(df_dyn_clean, Wp_dyn_clean, X_vars)

# 4) Tulosta tulokset
print(dyn_gmm.summary)
print("\n=== SHORT-RUN EFFECTS ===")
print(sr_effects)
print("\n=== LONG-RUN EFFECTS ===")
print(lr_effects)

from numpy.random import default_rng

def sar_slx_impacts(beta, theta, rho, W, R=200, seed=42):
    """
    SAR + SLX impacts (ADI / AII / ATI)
    LeSage & Pace (2009), Hutchinson trace approximation
    """
    rng = default_rng(seed)

    n = W.shape[0]
    I = eye(n, format="csc")
    A = I - rho * W
    lu = splu(A)

    ones = np.ones((n, 1))

    # --- TOTAL effect ---
    total = (
        lu.solve(beta * ones + theta * (W @ ones))
    ).mean()

    # --- DIRECT effect (trace approximation) ---
    Z = rng.choice([-1.0, 1.0], size=(n, R))   # R random vectors
    SZ = lu.solve(Z)
    trace_S = (Z * SZ).sum(axis=0).mean()     # Hutchinson trace
    trace_S /= n

    WZ = W @ Z
    SWZ = lu.solve(WZ)
    trace_SW = (Z * SWZ).sum(axis=0).mean()
    trace_SW /= n

    direct = beta * trace_S + theta * trace_SW
    indirect = total - direct

    return float(direct), float(indirect), float(total)

# --- 9.5 Common factor -koe SDM: SDM vs. SEM (Wald) --------------------------
# H0: theta_j + rho * beta_j = 0  for all regressors (SDM collapses to SEM)

from scipy.stats import chi2

def common_factor_wald(model, X_vars):
    """
    Wald-test for the common-factor restriction:
    H0: θ_j + ρ * β_j = 0  for all j in X_vars.
    Works across spreg versions where ρ may or may not be inside theta_hat.
    """
    # 1) Harmonized parameters and covariance
    theta_hat, cov, param_names = align_mean_cov(model)

    # 2) Build indexes for ρ, β and θ
    rho_idx, beta_idx, theta_idx = build_index_maps(param_names, X_vars)

    # 3) ρ-hat: if it's not in theta_hat-vector, take model.rho
    if rho_idx is None:
        rho_hat = float(model.rho)
    else:
        rho_hat = f

    # 4) Construct φ-vector and gradientmatrix
    p = cov.shape[0]
    phi_vals = []
    G = np.zeros((len(X_vars), p))

    for r, x in enumerate(X_vars):
        b_idx = beta_idx[x]
        t_idx = theta_idx[x]

        beta_hat = float(theta_hat[b_idx]) if b_idx is not None else 0.0
        theta_x  = float(theta_hat[t_idx]) if t_idx is not None else 0.0

        # φ_j = θ_j + ρ * β_j
        phi = theta_x + rho_hat * beta_hat
        phi_vals.append(phi)

        # gradients: ∂φ/∂ρ = β_j ; ∂φ/∂β_j = ρ ; ∂φ/∂θ_j = 1
        if rho_idx is not None:
            G[r, rho_idx] = beta_hat
        if b_idx is not None:
            G[r, b_idx] = rho_hat
        if t_idx is not None:
            G[r, t_idx] = 1.0

    phi = np.asarray(phi_vals).reshape(-1, 1)

    # 5) Var(φ) = G Σ G'
    V_phi = G @ cov @ G.T
    V_phi = (V_phi + V_phi.T) / 2.0
    V_phi += 1e-12 * np.eye(V_phi.shape[0])

    # 6) Wald = φ' V_phi^{-1} φ ~ χ^2_k,  k = len(X_vars)
    W_stat = float(phi.T @ np.linalg.inv(V_phi) @ phi)
    df = len(X_vars)
    p_val = 1.0 - chi2.cdf(W_stat, df)

    return W_stat, df, p_val, pd.Series(phi.flatten(), index=X_vars)


# ----- Call and save ----------------------------------------------------

# Use SDM-model, which includes WX-multipliers and covariance:
# m_sdm_imp (spat_impacts='all') on varmin valinta, mutta m_sdm_ml käy myös.
W_stat, df_cf, p_cf, phi_series = common_factor_wald(m_sdm_imp, X_vars)

print(f"Common factor Wald: W={W_stat:.3f}, df={df_cf}, p={p_cf:.4g}")

    f.write(f"Degrees of freedom: {df_cf}\n")
    f.write(f"p-value: {p_cf}\n")

def haversine_km(lon, lat):
    R=6371.0088
    lon=np.deg2rad(lon); lat=np.deg2rad(lat)
    N=len(lon); D=np.zeros((N,N), float)
    for i in range(N):
        dlon=lon-lon[i]; dlat=lat-lat[i]
        a=np.sin(dlat/2)**2 + np.cos(lat[i])*np.cos(lat)*np.sin(dlon/2)**2
        D[i,:]=2*R*np.arcsin(np.minimum(1.0, np.sqrt(a)))
    np.fill_diagonal(D,0.0); return D

def panelize_W(W_country, T):
    W_sparse=block_diag([W_country.sparse]*T, format="csr")
    row_ptr,col_ind,data=W_sparse.indptr,W_sparse.indices,W_sparse.data
    N=W_sparse.shape[0]; neighbors,weights={},{}
    for i in range(N):
        st,en=row_ptr[i],row_ptr[i+1]
        cols=col_ind[st:en]; vals=data[st:en]
        mask=cols!=i
        neighbors[i]=cols[mask].tolist(); weights[i]=vals[mask].tolist()
    Wp=W(neighbors=neighbors, weights=weights); Wp.transform="R"; return Wp

def W_country_knn(coords3, k, row_standardize=True):
    Wc = KNN.from_array(coords3, k=k)
    Wc.transform = "R" if row_standardize else "B"
    return Wc

def W_country_threshold(Dkm, d_max_km, row_standardize=True, ensure_min_neighbor=True):
    N=Dkm.shape[0]; neighbors={}; weights={}
    for i in range(N):
        idx=np.where((Dkm[i,:]>0)&(Dkm[i,:]<=d_max_km))[0]
        if ensure_min_neighbor and idx.size==0:
            j=np.argsort(Dkm[i,:])[1]; idx=np.array([j],dtype=int)
        neighbors[i]=idx.tolist(); weights[i]=[1.0]*len(idx)
    Wc=W(neighbors=neighbors, weights=weights)
    Wc.transform="R" if row_standardize else "B"; return Wc

def W_country_decay(Dkm, mode="inverse", alpha=1.0, half_life_km=None,
                    cutoff_km=None, kmax=None, row_standardize=True):
    if mode=="inverse":
        with np.errstate(divide='ignore'):
            Wraw=np.where(Dkm>0, Dkm**(-alpha), 0.0)
    elif mode=="exp":
        assert half_life_km and half_life_km>0
        lam=log(2.0)/float(half_life_km)
        Wraw=np.exp(-lam*Dkm); np.fill_diagonal(Wraw,0.0)
    else:
        raise ValueError("mode must be 'inverse' or 'exp'")
    if cutoff_km is not None:
        Wraw[Dkm>float(cutoff_km)]=0.0
    neighbors,weights={},{}
    for i in range(Wraw.shape[0]):
        wrow=Wraw[i,:].copy()
        if kmax is not None:
            nz=np.where(wrow>0)[0]
            if nz.size>kmax:
                topk = nz[np.argsort(wrow[nz])[::-1][:kmax]]
                keep = np.zeros_like(wrow,dtype=bool); keep[topk]=True
                wrow[~keep]=0.0
        idx=np.where(wrow>0)[0]
        neighbors[i]=idx.tolist(); weights[i]=wrow[idx].tolist()
    Wc=W(neighbors=neighbors, weights=weights)
    Wc.transform="R" if row_standardize else "B"; return Wc

# -- Safe SDM-run without impacts (to avoid power expansion problems) ---
def run_sdm_with_W(y, X, W_panel, X_vars):
    m = ML_Lag(
        y=y, x=X, w=W_panel,
        slx_lags=1, slx_vars="All",      # SDM
        method="full",
        spat_impacts=None,               # <-- EI impactsia gridissä
        spat_diag=False,                 # nopeampi ja vakaampi tässä vaiheessa
        name_y="Deforestation", name_x=X_vars, name_w="W_panel"
    )
    out = {
        "rho": float(m.rho),
        "aic": float(getattr(m, "aic", np.nan)),
        "loglik": float(getattr(m, "logll", np.nan)),
        "pseudoR2": float(getattr(m, "pr2_e", np.nan)),
        "n": int(m.n)
    }
    return m, out

def within_demean_country(arr, df, group_col="Country"):
    """
    arr : ndarray (N x k)
    df  : dataframe aligned with arr
    """
    out = arr.copy()
    for g, idx in df.groupby(group_col).indices.items():
        out[idx] -= out[idx].mean(axis=0)
    return out

def fmt_s8(center, lo, hi, dec=2):
    return (
        f"{center:.{dec}f}\n"
        f"({lo:.{dec}f}, {hi:.{dec}f})"
    )


import pandas as pd

def build_table_S8(eff_no, eff_year, eff_twfe):
    """
    Palauttaa DataFrame, joka vastaa Supplementin Table S8:a
    (ADI / AII / ATI, NO FE vs Year FE vs TWFE)
    """

    frames = {
        "NO FE": eff_no,
        "Year FE": eff_year,
        "TWFE": eff_twfe
    }

    rows = []

    variables = eff_no["Variable"].tolist()

    for var in variables:
        row = {"Variable": var}

        for model, eff in frames.items():
            r = eff.loc[eff["Variable"] == var].iloc[0]

            row[(model, "Direct")] = fmt_s8(
                r["Direct"], r["Direct_CI_L"], r["Direct_CI_U"]
            )
            row[(model, "Indirect")] = fmt_s8(
                r["Indirect"], r["Indirect_CI_L"], r["Indirect_CI_U"]
            )
            row[(model, "Total")] = fmt_s8(
                r["Total"], r["Total_CI_L"], r["Total_CI_U"]
            )

        rows.append(row)

    table = pd.DataFrame(rows)
    table = table.set_index("Variable")
    table.columns = pd.MultiIndex.from_tuples(
        table.columns, names=["Model", "Effect"]
    )

    return table


table_S8 = build_table_S8(eff_no, eff_year, eff_twfe)

import numpy as np
import pandas as pd
from scipy.stats import norm

def extract_standardized_safe(m, y_vec, X, label):
    """
    Standardized coefficients β* for BASE / LOGY / PERKM2.
    This version avoids ALL shape problems:
    - Standardizes only the ORIGINAL X_vars (not WX terms)
    - Uses beta_j only for matching original X_vars
    """
    names = list(m.name_x)

    # Only keep coefficients for the ORIGINAL X variables
    mask = [v in X_vars for v in names]   # X_vars is your original list

    betas = np.asarray(m.betas).flatten()[mask]
    names_keep = [nm for nm, keep in zip(names, mask) if keep]

    # sd(X_j) from your original X
    sd_x = X.std(axis=0)

    # sd(Y) from model's dependent vector
    sd_y = np.asarray(m.y).std(ddof=1)

    beta_std = betas * (sd_x / sd_y)

    # z statistics → p-values
    z_vals = []
    for row in m.z_stat:
        if len(row) >= 2:
            z_vals.append(float(row[1]))
        else:
            z_vals.append(np.nan)
    z_vals = np.asarray(z_vals)[mask]
    pvals = 2*(1 - norm.cdf(np.abs(z_vals)))

    return pd.DataFrame({
        "Variable": names_keep,
        f"{label}_beta_std": beta_std,
        f"{label}_p": pvals
    })
